# This project can run three different related file management apps
#
#   The three apps are:
#       * Fingerprint Database Maker
#       * Unique File Copier
#		* Photo Collector and Organizer

# Important: keep WhichApp the first category and setting
# Possible Apps: FingerprintDBMaker, UniqueFileCopier, PhotoCollector
#
# If the app is named one of these three names then the app name will override the setting below
#
[WhichApp]
# FingerprintDBMaker
UniqueFileCopier
# PhotoCollector



#-----------------------------------------------------------------------------------------------------------------------------------------
# C# types for all Key = Value settings below:
# bool setting names start with '?', int setting names start with '+', all other settings are strings
#-----------------------------------------------------------------------------------------------------------------------------------------

[All Apps]
?ExcludeHiddenAndSystem = True

DefaultDestinationVolume = C:

# Other settings files
CategoriesConfigFileName	= FileCategoriesByExtension.txt
DirectoriesIgnoreFileName	= DirectoriesIgnore.txt
FilesIgnoreFileName			= FilesIgnoreByExtension.txt

# Report and DB files
ReportFilesSubdirectory		= New Reports
ReportFilesExtension		= .tsv
FilesDBNameSuffix			= -File DB
ExcludedReportNameSuffix	= -Excluded List
DuplicatesReportNameSuffix	= -Duplicate Files
UniquesReportNameSuffix		= -Unique Files

unknownCategoryName			= Unknown

DBFileNameFilter			= *-File DB.tsv			# Match pattern above

# Category names in settings files
DirectoriesIgnoreCategory   = Directories Ignore
ExtensionsIgnoreCategory    = Extensions Ignore

PhotoFileExtensionsCategory = Photo
VideoFileExtensionsCategory = Video

#-----------------------------------------------------------------------------------------------------------------------------------------

[FingerprintDBMaker]
appName					= Fingerprint Database Maker
appDescription			= Fingerprints unique files in the target volume/directory and creates a DB file in the New Reports directory
operationDescription	= Fingerprinting unique files from

BaseFileDBsDirectory	= Base File DBs

#-----------------------------------------------------------------------------------------------------------------------------------------

[UniqueFileCopier]
appName					= Unique File Copier
appDescription			= Copies unique files found in the target volume/directory and optionally organizes them into folders by file category
operationDescription	= Copying unique files from

BaseFileDBsDirectory	= Base File DBs

destRootPrefix			= unq-

#-----------------------------------------------------------------------------------------------------------------------------------------

[PhotoCollector]
appName					= Photo Collector and Organizer
appDescription			= Collects unique photos and videos and organizes them into folders by year taken
operationDescription	= Collecting and organizing unique photos and videos from

photosDestRootFolder	= Collected Photos
videosDestRootFolder	= Collected Videos

BaseFileDBsDirectory	= Photos File DBs

unknownYearName			= Unknown Year
